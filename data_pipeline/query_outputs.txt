Query 6 - JOIN Books and Categories

            b.book_id,
            b.title,
            b.price_gbp,
            b.price_inr,
            b.rating,
            b.in_stock,
            c.category_name
        FROM books b
        INNER JOIN categories c
            ON b.category_id = c.category_id
